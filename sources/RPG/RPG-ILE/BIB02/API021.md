# BIB02/QRPGLESRC/API021

```rpg
**free
ctl-opt dftactgrp(*no) actgrp(*caller);

dcl-s idtoken char(108);
dcl-s wkctr char(10);
dcl-s wcdsc char(50);
// dcl-s jsonStr sqltype(clob:32000);
dcl-s jsonStr varchar(32000);
dcl-s WJson varchar(10000);
dcl-s wrc varchar(25500);
dcl-s apiUrl varchar(256) inz('https://gorest.co.in/public/v2/users');
dcl-s httpHeaders varchar(10000);

idtoken = 'd31eda74376450fd9a2d74c873c5095aeca6a85d97e6561f85671a4ab8498da2';

EXEC SQL
  SET option commit = *none , ALWCPYDTA = *OPTIMIZE, CLOSQLCSR = *ENDMOD;

exec sql
    declare c1 cursor for
        select WKCTR, WCDSC
        from WRKCTR;

exec sql
    open c1;

dow sqlcod = 0;
    exec sql
        fetch next from c1 into :wkctr, :wcdsc;

    if sqlcod = 0;
        if %len(%trim(WJson)) > 18;
            WJson += ',';
        endif;
        WJson += '{ "name": "' + %trim(wkctr) + '", "email": "' + %trim(wcdsc)
        + '", "gender": "male" , "status": "active" }';
    endif;
enddo;

exec sql
    close c1;
JsonStr = %trim(Wjson);
httpHeaders = '<httpheader><header name=';
httpHeaders += '"Content-Type" value="application/json"></header>';
httpHeaders += '<header name="Authorization" value="Bearer ';
httpHeaders += %trim(Idtoken) + '" ></header>';
httpHeaders += '</httpheader>';

Exec sql
    values systools.HTTPPOSTCLOB(:apiUrl, :httpHeaders, :JsonStr)
    into :wrc;
if sqlstate = '00000';
    dsply 'Charge posts sent successfully!';
else;
    dsply 'Error sending charge posts:';
endif;

*inlr = *on;
```
