# BIB02/QCHATGPT/T3

```rpg
**FREE
ctl-opt dftactgrp(*no) actgrp(*caller);

dcl-s vidmart zoned(5:0);
dcl-s vnomart char(20);
dcl-s vprix   zoned(7:2);
dcl-s msg char(52);

exec sql
set option commit = *none;

exec sql
declare c1 cursor for
select idart, nomart, prix
from articlepf;


exec sql
open C1;

exec sql
fetch c1 into :vidmart, :vnomart, :vprix;

dow sqlcode = 0;

  if vprix >= 50;
    msg = 'Prix : ' + %char(vprix);
  else;
    msg = 'Prix trop bas';
  endif;

  dsply msg;

  exec sql
    fetch c1 into :vidmart, :vnomart, :vprix;
enddo;

*inlr = *on;
```
