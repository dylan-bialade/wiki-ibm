# BIB02/QRPGLESRC/API020

```rpg
**free
ctl-opt dftactgrp(*no) actgrp(*caller);
dcl-pi *n;
     Idtoken char(48);
end-pi;
dcl-s wkctr char(10);
dcl-s wcdsc char(50);
dcl-s jsonStr sqltype(clob:16000000);
dcl-s WJson varchar(20000);
dcl-s wrc varchar(25500);
dcl-s apiUrl varchar(256) inz('https://votre-api.com/api/chargeposts');
dcl-s httpHeaders varchar(20000);

exec sql
    declare c1 cursor for
        select WKCTR, WCDSC
        from WRKCTR;

exec sql
    open c1;
WJson = '{ "chargeposts": [';

dow sqlcod = 0;
    exec sql
        fetch next from c1 into :wkctr, :wcdsc;

    if sqlcod = 0;
        if %len(%trim(WJson)) > 18; // 18 because '{ "chargeposts": [' length is 18
            WJson += ',';
        endif;
        WJson += '{ "wkctr": "' + %trim(wkctr) + '", "wcdsc": "' + %trim(wcdsc) + '" }]}';
    endif;
enddo;

exec sql
    close c1;
JsonStr_data = %trim(Wjson);
httpHeaders = '<httpheader><header name=';
httpHeaders += '"Content-Type" value="application/json"></header>';
httpHeaders += '"Authorization"' + 'value="';
httpHeaders += %trim(Idtoken) + '" ></header>';
httpHeaders += '</httpheader>';

Exec sql
    values systools.HTTPPOSTCLOB(:apiUrl, :httpHeaders, :JsonStr)
    into :wrc;
if sqlstate = '00000';
    dsply 'Charge posts sent successfully!';
else;
    dsply 'Error sending charge posts:';
endif;

*inlr = *on;
```
