      //--------------------------------------------------------*

     FF52W1FM   cf   e             workstn
     FSALARF52  uf a e           k disk

     DF52W1            PR                  extpgm('F52W1')
     D P_SACOD                             options(*NOPASS)
     D                                     like(SACOD)
      *
     DF52W1            PI
     D P_SACOD                             options(*NOPASS)
     D                                     like(SACOD)

     D SACODW          s                   like(SACOD)
     D SACODN          s              3  0

      //* Lecture fiche
      /FREE
       If %PARMS > 0;
       Chain(n) P_SACOD SALARF;
       ENDIF;
       //
       //* Affichage
       DoW Not *InLR;
         Exfmt F1;
         If *InKC;
           *InLR = *On;
         Else;
           Chain SACOD SALARF;
           If Not %Found;
             SetGT *HiVal SALARF;
             ReadP SALARF;
             SACODN = %Dec(SACOD:3:0) + 1;
             SACODW = %EditC(SACODN:'X');
             Read F1; // Relecture écran avant ajout
             SACOD = SACODW;
             Write SALARF;
           Else;
            Read F1;  // Relecture écran avant màj
            Update SALARF;
           EndIf;
           *InLR = *On;
         EndIf;
       EndDo;
       //
      /END-FREE
