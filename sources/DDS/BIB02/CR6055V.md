# BIB02/QDDSRC/CR6055V

```text
     H*                                                                   CR6055
     H DECEDIT('0,') DFTNAME(CR6055) DATEDIT(*DMY.) DFTACTGRP(*NO)
     F********************************************************************
     F* AUTO CR6055 - MODIFICATION DU REGIME                             *
     F********************************************************************
       DCL-F CR6055FM WORKSTN;
       DCL-F CPTE01L USAGE(*INPUT) KEYED;
       DCL-F CLI01L USAGE(*INPUT) KEYED;

       DCL-S TM ZONED(2) DIM(12) CTDATA PERRCD(12); // JOUR PAR MOIS
       DCL-S LC ZONED(4) DIM(9) CTDATA PERRCD(1);

       DCL-S MES CHAR(76) DIM(9) ALT(LC);
       DCL-S ERL CHAR(1) DIM(85);
       DCL-S CLE6 ZONED(6);
       DCL-S CLE1 ZONED(1);
       DCL-S K ZONED(1);
       DCL-S DATE1 ZONED(4);
       DCL-S DATE2 ZONED(4);
       DCL-S DATE8 ZONED(8);
       DCL-S REALDATE DATE;
       DCL-S ZONE8 ZONED(8);
       DCL-S D11 ZONED(2);
       DCL-S D11B ZONED(4);
       DCL-S D12 ZONED(2);
       DCL-S D12B ZONED(4);
       DCL-S CHAR2 CHAR(2);
       DCL-S CHAR4 CHAR(4);
       DCL-S CHAR8 CHAR(8);



       //DCL-DS *N DTAARA(*LDA);
       //ECOTIS PACKED(6);
       //EDATE PACKED(8);
       //END-DS;

       DCL-DS *N;
         CLECPT CHAR(16);
         CLECLI ZONED(6);
         CLECAI ZONED(2);
         CLEDTV ZONED(8);
       END-DS;

       DCL-DS *N;
         COSSVE ZONED(2);
         COAAVE ZONED(2);
         COMMVE ZONED(2);
         COJJVE ZONED(2);
         COSSAA ZONED(4) SAMEPOS(COSSVE);
         CODTV ZONED(8) SAMEPOS(COSSVE);
       END-DS;

       //*INLR = *ON;
       *IN07 =*OFF;
       *IN01 = *OFF;
       *IN02 = *OFF;

       DOU *IN07;
         SELECT;
         WHEN *IN07;
           LEAVE;
         OTHER;

           IF *IN81 = *ON;
             EXFMT FORMAT01;
           //*IN01=*ON;
           ELSEIF *IN81 = *OFF;
             *IN81 = *ON;

           ENDIF;

           // INDICATEURS :
           // 01 : Affiche écran 1 ( demande n° cotis. et date )
           // 02 : Affiche écran 2 ( Confirme, passage TR suivant ou sortie )
           // 10 : Contrôle de passage Prem. lecture de CPTE
           // 99 : Affiche messages d'erreur
           // U1 : Confirme et passage traitement suivant : CR6050

           *IN99 = *OFF;

           //    IF ECOTIS = ' ';
           //      *IN01 = *OFF;
           //    ELSE;
           //      *IN01 = *ON;
           //    ENDIF;
           //
           //    IF EDATE = ' ';
           //      *IN02 = *OFF;
           //    ELSE;
           //      *IN02 = *ON;
           //    ENDIF;


           IF *IN01 = *OFF AND *IN02 = *OFF;
             TR00();

           ELSEIF *IN01;
             TR01();

           ENDIF;

           IF *IN02;
             TR02();

           ENDIF;

         ENDSL;

       ENDDO;



       // DCL-PROC ********************************************************************************

       // Traitement écran de départ à blanc
       DCL-PROC TR00;

         *INU1 = *OFF;
         ECOTIS = *BLANK;
         EDATE = *BLANK;
         WRITE FORMAT01;
         *IN01 = *ON;
       END-PROC;


       // Traitement écran 1
       DCL-PROC TR01;
         IF *INKG;
           *INLR = *ON;
         ELSE;
           CLE6 = %DEC(ECOTIS:6:0);
           CLE1 = 1;
           CHAIN (CLE6:CLE1) CLI01L;
           IF NOT %FOUND;
             //*IN99 = *ON;
             EERREU = 'COTISANT INCONNU';
             EXFMT ERREUR;
           ELSE;

             IF CLREGI = 'A';
               EERREU = 'ALIENATION TOTALE IMPOSSIBLE,+
                LE REGIME EST DEJA ALIENE';
               EXFMT ERREUR;
             ELSEIF CLREGI = 'T';
               K = 3;
               EERREU = 'ALIENATION TOTALE IMPOSSIBLE,+
                LE REGIME EST RESERVE TEMPORAIRE';
               EXFMT ERREUR;
             ELSE;
               CTDATE();
               CTDTSP();
               WRITE FORMAT02;
               *IN02 = *ON;
             ENDIF;

           ENDIF;

         ENDIF;

       END-PROC;

       // Traitement écran 2
       DCL-PROC TR02;
         IF *INKG = *OFF AND *INKJ = *OFF;
           WRITE FORMAT02;
           *IN02 = *ON;
         ELSEIF *INKG;
           *INLR = *ON;

         ELSEIF *INKJ;
           *INLR = *ON;
           *INU1 = *ON;

         ENDIF;
       END-PROC;

       // Contrôle de date
       DCL-PROC CTDATE;
         DATE1 = %DEC(%SUBST(EDATE:1:4):4:0);
         DATE2 = %DEC(%SUBST(EDATE:5):4:0);
         D11 = %DEC(%SUBST(%CHAR(DATE1):1:2):2:0);
         D12 = %DEC(%SUBST(%CHAR(DATE1):3):2:0);

         IF D12 > 12 OR D12 < 1;
           //K = 4;
           *IN99 = *ON;
           EERREU = 'DATE INVALIDE';
           EXFMT ERREUR;
         ELSEIF D11 > TM(D12) ;
           //K = 4;
           *IN99 = *ON;
           EERREU = 'DATE INVALIDE';
           EXFMT ERREUR;
         ELSE;
           CHAR8 = %CHAR(*DATE);
           D11B = %DEC(%SUBST(CHAR8:5):4:0);
           D12B = DATE2;
           IF D12B > D11B;
             //K = 5;
             *IN99 = *ON;
             EERREU = 'ANNEE SAISIE DOIT ETRE : +
             ANNEE EN COURS OU ANNEE ( N - 1 )';
             EXFMT ERREUR;
           ELSE;
             D11B = D11B - 1;
             IF D12B < D11B;
               //K = 5;
               *IN99 = *ON;
               EERREU = 'ANNEE SAISIE DOIT ETRE : +
               ANNEE EN COURS OU ANNEE ( N - 1 )';
               EXFMT ERREUR;
             ELSE;
               CHAR2 = %SUBST(%CHAR(DATE2):1:2);
               D11 = %DEC(CHAR2:2:0);
               IF D11 > 20 OR D11 < 19;
                 //K = 4;
                 *IN99 = *ON;
                 EERREU = 'DATE INVALIDE';
                 EXFMT ERREUR;
               ENDIF;

             ENDIF;

           ENDIF;

         ENDIF;

       END-PROC;

       // Contrôle date saisie : >= date versement du dernier mouvement
       DCL-PROC CTDTSP;
         // Préparation de la date saisie : JJMMAAAA devient AAAAMMJJ
         ZONE8 = %DEC(EDATE:8:0);
         REALDATE = %DATE(ZONE8:*EUR);
         DATE8 =  %DEC(REALDATE:*ISO);
         *IN10 = *ON;
         CLECAI = 0;
         CLEDTV = 0;
         CLECLI = %DEC(ECOTIS:6:0);
         SETLL CLECLI CPTE01L;
         // TAG LECT
         READ CPTE01L;
         DOU %EOF;

           IF %EOF;
             *IN99 = *ON;
           ENDIF;

           IF *IN99;

             IF *IN10;
               *IN10 = *OFF;
               //K = 7;
               EERREU = 'IL N''Y A AUCUN MOUVEMENT +
                POUR CE COTISANT DANS FIC. CPTE';
               EXFMT ERREUR;
               LEAVE;

             ELSE;
               *IN99 = *OFF;
               *IN10 = *OFF;
               LEAVE;
             ENDIF;

           ENDIF;

           *IN10 = *OFF;

           IF COCLI <> %DEC(ECOTIS:6:0);
             *IN99 = *OFF;
             LEAVE;
           ENDIF;

           IF *IN99 = *OFF;
             IF DATE8 <= CODTV;
               //K = 6;
               EERREU = 'IL RESTE DES MOUVEMENTS DANS FIC.CPTE +
               AYANT UNE DATE VERS.> DATE SAISIE';
               EXFMT ERREUR;
               LEAVE;
             ENDIF;
           ENDIF;

           READ CPTE01L;

         ENDDO;

       END-PROC;

       // END DCL-PROC ****************************************************************************
```
