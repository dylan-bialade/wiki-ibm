# BIALADE/QCLPSRC/IMPOCL

```cl
PGM

    DCL VAR(&NOMDTA) TYPE(*CHAR) LEN(10) VALUE(REVDTA)

    /* CRÉE LA DTAARA SEULEMENT SI ELLE N'EXISTE PAS */
    CHKOBJ OBJ(BIALADE/REVDTA) OBJTYPE(*DTAARA)
    MONMSG MSGID(CPF9801) EXEC(DO)
        CRTDTAARA DTAARA(BIALADE/REVDTA) TYPE(*DEC) LEN(10 0) +
                    VALUE(123456)
    ENDDO

    /* PASSE LE NOM DE LA DTAARA AU RPG */
    CALL PGM(BIALADE/IMPORP) PARM(&NOMDTA)

ENDPGM
```
